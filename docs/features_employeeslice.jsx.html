/**
 * Redux slice for managing employee-related state.
 * 
 * This slice includes actions and reducers for adding employees, setting a specific employee,
 * clearing the selected employee, and loading employees from an external source (e.g., localStorage).
 * 
 * @module employeeSlice
 */

import { createSlice } from "@reduxjs/toolkit";

/**
 * Initial state for the employee slice.
 * 
 * @typedef {Object} EmployeeState
 * @property {Array<Object>} employees - List of all employees.
 * @property {Object} employee - Currently selected employee.
 */
const initialState = {
    employees: [],
    employee: {},
};

/**
 * Redux slice for employee management.
 */
const employeeSlice = createSlice({
    name: "employee",
    initialState,
    reducers: {
        /**
         * Adds a new employee to the list of employees.
         * 
         * @function
         * @param {EmployeeState} state - Current state of the slice.
         * @param {Object} action - Redux action.
         * @param {Object} action.payload - New employee to add.
         */
        addEmployee: (state, action) => {
            state.employees = [...state.employees, action.payload];
        },

        /**
         * Sets the currently selected employee.
         * 
         * @function
         * @param {EmployeeState} state - Current state of the slice.
         * @param {Object} action - Redux action.
         * @param {Object} action.payload - Employee data to set.
         */
        setEmployee: (state, action) => {
            state.employee = action.payload;
        },

        /**
         * Clears the currently selected employee.
         * 
         * @function
         * @param {EmployeeState} state - Current state of the slice.
         */
        clearEmployee: (state) => {
            state.employee = {};
        },

        /**
         * Loads a list of employees into the state.
         * 
         * @function
         * @param {EmployeeState} state - Current state of the slice.
         * @param {Object} action - Redux action.
         * @param {Array<Object>} action.payload - List of employees to load.
         */
        loadEmployees: (state, action) => {
            state.employees = action.payload;
        },
    },
});

/**
 * Action creators generated by createSlice.
 * 
 * @typedef {Object} EmployeeActions
 * @property {Function} addEmployee - Action to add a new employee.
 * @property {Function} setEmployee - Action to set the selected employee.
 * @property {Function} clearEmployee - Action to clear the selected employee.
 * @property {Function} loadEmployees - Action to load employees into the state.
 */
export const { addEmployee, setEmployee, clearEmployee, loadEmployees } = employeeSlice.actions;

/**
 * Selector to get the list of employees from the state.
 * 
 * @function
 * @param {Object} state - Redux state.
 * @returns {Array<Object>} List of employees.
 */
export const selectEmployees = (state) => state.employee.employees;

/**
 * Selector to get the currently selected employee from the state.
 * 
 * @function
 * @param {Object} state - Redux state.
 * @returns {Object} Currently selected employee.
 */
export const selectEmployee = (state) => state.employee.employee;

export default employeeSlice.reducer;
