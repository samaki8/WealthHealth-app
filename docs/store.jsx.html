/**
 * Configures and creates a Redux store with persistence capabilities.
 * 
 * This store uses Redux Toolkit's `configureStore` to set up the store
 * and integrates Redux Persist to persist the state of the `employee` slice
 * in local storage. The persisted state is rehydrated when the application loads.
 * 
 * Key Features:
 * - Combines reducers using `combineReducers` (currently only `employeeSlice`).
 * - Persists the `employee` slice using Redux Persist.
 * - Configures middleware to ignore serialization checks for Redux Persist actions.
 * 
 * @module store
 * 
 * @requires @reduxjs/toolkit
 * @requires redux-persist
 * @requires redux-persist/lib/storage
 * @requires redux
 * @requires ./features/employeeslice
 * 
 * @constant {Object} persistConfig - Configuration for Redux Persist.
 * @property {string} persistConfig.key - The key for the persisted state.
 * @property {Object} persistConfig.storage - The storage engine (local storage).
 * @property {string[]} persistConfig.whitelist - List of reducers to persist.
 * 
 * @constant {Function} rootReducer - Combines all reducers into a single reducer.
 * 
 * @constant {Function} persistedReducer - Enhances the root reducer with persistence capabilities.
 * 
 * @constant {Object} store - The configured Redux store.
 * @property {Function} store.dispatch - Dispatches actions to the store.
 * @property {Function} store.getState - Retrieves the current state of the store.
 * @property {Function} store.subscribe - Subscribes to store updates.
 * 
 * @constant {Object} persistor - The persistor object for managing persistence.
 * @property {Function} persistor.persist - Starts persisting the store.
 * @property {Function} persistor.flush - Flushes all pending state writes.
 * 
 * @exports store - The configured Redux store.
 * @exports persistor - The persistor object for managing persistence.
 */
// WealthHealth-app\src\store.jsx

import { configureStore } from '@reduxjs/toolkit';
import { persistStore, persistReducer } from 'redux-persist';
import storage from 'redux-persist/lib/storage';
import { combineReducers } from 'redux'; // Ajoutez ceci
import employeeSlice from './features/employeeslice';

// 1. Combinez les reducers même s'il n'y en a qu'un
const rootReducer = combineReducers({
    employee: employeeSlice,
});

const persistConfig = {
    key: 'root',
    storage,
    whitelist: ['employee'],
};

// 2. Utilisez rootReducer comme base
const persistedReducer = persistReducer(persistConfig, rootReducer);

// 3. Créez le store avec le reducer persisté
const store = configureStore({
    reducer: persistedReducer,
    middleware: (getDefaultMiddleware) =>
        getDefaultMiddleware({
            serializableCheck: {
                ignoredActions: ['persist/PERSIST'],
            },
        }),
});

export const persistor = persistStore(store);
export default store;


{/* Travailler sur la bibliothèque persistante Redux
Sérialisation : chaque fois que l'état de Redux change, Redux Persist intercepte l'état de votre magasin Redux et le convertit dans un format adapté au stockage comme une chaîne JSON .
État du magasin : après la sérialisation, Redux Persist enregistre ces données dans un stockage spécifié (par exemple, un stockage local).
Réhydratation : lorsque l'application se charge, Redux Persist récupère l'état persistant du stockage et le réinjecte dans le magasin Redux avant de restituer l'interface utilisateur.*/}
